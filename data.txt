SELECT
      SELECT DISTINCT
       T2.ICI,
       T5.HOH_UPI,
       T5.APPL_UPI

FROM
       W100DP1.TWNELIG_VERSION     T2,
       W100DP1.TWNELG_VER_MBR      T3,
       W100DP1.TWNCASE_MEMBER_HST  T4,
       W100DP1.TWNFAM_CASE_HSTRY   T5

WHERE
      
       T2.PRGM_CASE_TYPE   = 'AF'
   AND T2.BNFT_YR          = YEAR(:RUN_DT)
   AND T2.BNFT_MTH         = MONTH(:RUN_DT)
   AND T2.CASE_RSLT_CD     = 'E'
   AND T2.LST_POSTD_VER_IND = 'Y'
   AND T3.ICI              = T2.ICI
   AND T3.PRGM_CASE_TYPE   = T2.PRGM_CASE_TYPE
   AND T3.BNFT_YR          = T2.BNFT_YR
   AND T3.BNFT_MTH         = T2.BNFT_MTH
   AND T3.VER_NMB          = T2.VER_NMB
   AND T3.AID_CD IN ('CA', 'AM1', 'EM2')
   AND T3.ELIG_CD NOT IN (' ', '3', '7', '9')
   AND T4.ICI              = T3.ICI
   AND T4.UPI              = T3.UPI
   AND T4.PRD_BEG_DT      <= :RUN_DT
   AND T4.PRD_END_DT      >= LAST_DAY(:RUN_DT)
   AND T4.MAINT_CD        <> 'D'
   AND T5.ICI              = T2.ICI
   AND T5.PRD_BEG_DT      <= :RUN_DT
   AND T5.PRD_END_DT      >= LAST_DAY(:RUN_DT)

GROUP BY
       T2.ICI

ORDER BY
       T2.ICI

FOR FETCH ONLY WITH UR;
