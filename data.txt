	/**
	 * WR #: BRGUS00201531 Author : Kumar Karuppanan Method : addDecimalAmt
	 * Description : To add the decimal point for the amount
	 * 
	 * @param amount
	 * @return formattd String for the amount
	 */

	private String addDecimalAmt(String amount) {
		if (amount != null && amount.trim().length() > 0
				&& !amount.trim().equals("0")) {
			if (amount.indexOf(".") == -1)
				return (amount + ".00");
			else {
				String decimal = amount.substring(amount.indexOf(".") + 1);

				if (decimal.trim().length() <= 0) {
					decimal = "00";
				} else if (decimal.trim().length() == 1) {
					decimal = decimal + "0";
				} else if (decimal.trim().length() > 2) {
					decimal = decimal.substring(0, 2);
				}
				return (amount.substring(0, amount.indexOf(".")) + "." + decimal);
			}
		} else
			return "";
	}

	/**
	 * @param textCode
	 * @return
	 */
	public String replaceHTMLTags(String textCode) {

		textCode = textCode.replaceAll("<Strong>", "");
		textCode = textCode.replaceAll("</Strong>", "");
		textCode = textCode.replaceAll("<strong>", "");
		textCode = textCode.replaceAll("</strong>", "");
		textCode = textCode.replaceAll("<STRONG>", "");
		textCode = textCode.replaceAll("</STRONG>", "");
		textCode = textCode.replaceAll("<b>", "");
		textCode = textCode.replaceAll("</b>", "");
		textCode = textCode.replaceAll("<B>", "");
		textCode = textCode.replaceAll("</B>", "");
		textCode = textCode.replaceAll("<br>", "");
		textCode = textCode.replaceAll("</br>", "");
		textCode = textCode.replaceAll("<Br>", "");
		textCode = textCode.replaceAll("</Br>", "");
		textCode = textCode.replaceAll("<BR>", "");
		textCode = textCode.replaceAll("</BR>", "");
		return textCode;
	}
