	private boolean findValidBurialRecords(APP_IN_BURY_ASET_Collection burialAsetColl) {
		APP_IN_BURY_ASET_Cargo appInBuryAsetCargo = null;
		boolean hasValidRecords = false;
		for (int i = 0; i < burialAsetColl.size(); i++) {
			appInBuryAsetCargo = burialAsetColl.getCargo(i);
			if(!appInBuryAsetCargo.getSrc_app_ind().contains("CW")){
				hasValidRecords = true;
				break;
			}
		}
		return hasValidRecords;
	}
