	private Set loadPrograms(String caseNumber){
		FwDataCriteria[] criteria = new FwDataCriteria[1];
		criteria[0] = new FwDataCriteria();
		criteria[0].setColumn_value(caseNumber);
		criteria[0].setData_type(FwConstants.STRING);
		
		Map sqlMap = new HashMap();
		sqlMap.put(FwConstants.SQL_IND, "sql-PRGCASE");
		sqlMap.put(FwConstants.CRITERIA, criteria);
		
		IData data = (IData)FwServiceFactory.getInstance().create(IData.class);
		List res = data.execute(FwConstants.SQL, sqlMap);
		
		Set availableCaseProg=new HashSet<String>(); 
		if(res != null && !res.isEmpty()){
			for(int i = 0; i < res.size(); i++) {
				Map cargo = (Map) res.get(i);
				String progCd = (String) cargo.get("PROG_CD");
				availableCaseProg.add(progCd);
			}			
		}
		return availableCaseProg;
	}
