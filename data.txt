	/***
	 * Method added by kumar karuppanan to validate the date
	 * 
	 * @param sDate
	 *            string date
	 * @return returns is high date or not.
	 */
	private boolean isHighDate(String sDate) {
		if (sDate != null) {
			sDate = sDate.substring(0, 10);
			if (sDate.indexOf(" ") == -1
					&& sDate.equals(AppConstants.HIGH_DATE))
				return true;
			else if (sDate.indexOf(" ") != -1
					&& sDate.equals(AppConstants.HIGH_TIMESTAMP))
				return true;
		}
		return false;
	}
