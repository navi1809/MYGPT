private void prescriptionDrugSection(Document doc, Map PDFMap, Map namesMap, Map agesMap) {
		String indvSeqNum = null;
		String appNumber = null;
		String caseNumber = null;
		try {
			appNumber = (String) PDFMap.get(AppConstants.APP_NUMBER);
			caseNumber = (String) PDFMap.get(AppConstants.CASE_NUM);
			String langIndicator = (String) PDFMap.get(AppConstants.LANGUAGE);
			doc.add(new Paragraph("\n"));

			RMC_IN_PRFL_Collection rmcInPrflColl = (RMC_IN_PRFL_Collection) PDFMap.get(AppConstants.RMC_IN_PRFL_MASTER);
			PdfPTable tab = new PdfPTable(2);
			float width[] = { 1, 1 };
			tab.setWidths(width);
			tab.setTotalWidth(100);
			tab.setWidthPercentage(100);

			RMC_IN_PRFL_Cargo rmc_IN_PRFL_Cargo = null;
			for (int i = 0; i < rmcInPrflColl.size(); i++) {
				rmc_IN_PRFL_Cargo = (RMC_IN_PRFL_Cargo) rmcInPrflColl.getCargo(i);
				indvSeqNum = rmc_IN_PRFL_Cargo.getIndv_seq_num();
				String prsnName = (String) namesMap.get(indvSeqNum);
				Date dobDate = dateRoutine.getDateFromTimeStamp((String) agesMap.get(indvSeqNum));
				String prsnAge = String.valueOf(dateRoutine.getAge(dobDate, fwDate.getDate()).getYears());

				if (i == 0) {
					doc.add(new Paragraph("\n"));
					// Person name label
					PdfPCell lblPersonName = this.getPdfPCell(this.getDisplayText(40119, langIndicator), fRegularTextBold, langIndicator);
					errorSectionHeader =  this.getDisplayText( 40119, langIndicator);

					lblPersonName.setGrayFill(0.8f);
					tab.addCell(lblPersonName);

					// Drug/parole question
					PdfPCell paroleDesc = this.getPdfPCell(this.getDisplayText(78678, langIndicator), fRegularTextBold, langIndicator);
					errorSectionHeader =  this.getDisplayText( 78678, langIndicator);
					paroleDesc.setGrayFill(0.8f);
					tab.addCell(paroleDesc);
				}

				// Person Name value
				Paragraph pPersonName = new Paragraph(AppConstants.EMPTY_STRING, fRegularTextTen);
				if ((prsnName != null) && (prsnName.length() != 0) && (!prsnName.equals(FwConstants.SPACE))) {
					pPersonName.add(prsnName);
					pPersonName.add("\n");
					pPersonName.add((String) this.getDisplayText(30660, langIndicator));
					errorSectionHeader =  this.getDisplayText( 30660, langIndicator);
					pPersonName.add(prsnAge);
				} else {
					pPersonName.add(FwConstants.SPACE);
				}

				PdfPCell personName = this.getPdfPCellFromParagraph(pPersonName);
				tab.addCell(personName);

				// Drug/parole response
				Paragraph pparoleDetail = new Paragraph(AppConstants.SPACE, fRegularTextTen);
				String paroleDetail = FwConstants.SPACE;
				String drugFelnResp = rmc_IN_PRFL_Cargo.getPrescription_drug_sw();
				if (drugFelnResp != null && drugFelnResp.toString().equalsIgnoreCase("R")) {
					paroleDetail = (String) this.getColumnValue("TYRN", 745, "Y", langIndicator);
				} else
					paroleDetail = (String) this.getColumnValue("TYRN", 745, "N", langIndicator);

				pparoleDetail.add(paroleDetail);
				PdfPCell paroleCell = this.getPdfPCellFromParagraph(pparoleDetail);
				tab.addCell(paroleCell);

			}
			doc.add(tab);

		} catch (DocumentException de) {
			StringWriter sw = new StringWriter();
			sw.append("prescriptionDrugSectionPDF").append("/appNumber:").append(appNumber).append("/indvSeqNum:").append(indvSeqNum).append("/caseNumber:").append(caseNumber).append("/");
			logger.log(this.getClass(), ILog.ERROR, sw.toString());

			FwException fe = new FwException();
			fe.setClassID(this.getClass().getName());
			fe.setMethodID("drugFelRespSection");
			fe.setExceptionType(FwConstants.EXP_TYP_APPLICATION);
			fe.setExceptionText(String.valueOf(de));
			fe.setStackTraceValue(FwExceptionManager.getStackTraceValue(de));
			writeExceptionRecord(fe); 
 //throw fe;;
		} catch (Exception e) {
			StringWriter sw = new StringWriter();
			sw.append("prescriptionDrugSectionPDF").append("/appNumber:").append(appNumber).append("/indvSeqNum:").append(indvSeqNum).append("/caseNumber:").append(caseNumber).append("/");
			logger.log(this.getClass(), ILog.ERROR, sw.toString());

			FwException fe = new FwException();
			fe.setClassID(this.getClass().getName());
			fe.setMethodID("drugFelRespSection");
			fe.setExceptionType(FwConstants.EXP_TYP_APPLICATION);
			fe.setExceptionText(String.valueOf(e));
			fe.setStackTraceValue(FwExceptionManager.getStackTraceValue(e));
			writeExceptionRecord(fe); 
 //throw fe;;
		}
	}

	private PdfPCell prepareProgramSelectedForPDFOnly(RMC_IN_PRFL_Cargo rmcInPrflCargo, String langIndicator, ArrayList<String> previousProgramCodes) throws Exception
	{
		// Build paragraph with SNAP
		Paragraph programParagraph = new Paragraph(this.getDisplayText(32114, langIndicator), fRegularTextTen);
		programParagraph.setAlignment(Paragraph.ALIGN_RIGHT);

		PdfPCell cell = this.getPdfPCell(programParagraph, langIndicator);
		cell.setHorizontalAlignment(PdfPCell.ALIGN_RIGHT);
		cell.setColspan(4);

		final String ONE = "1";
		// SNAP
		if (AppConstants.ZERO.equals(rmcInPrflCargo.getIndv_fs_rqst_ind()))
		{
			pdfCheckboxCheckNo(programParagraph, langIndicator);
		}
		else if(ONE.equalsIgnoreCase(rmcInPrflCargo.getIndv_fs_rqst_ind()))
		{
			pdfCheckboxCheckYes(programParagraph, langIndicator);
		}
		else
		{
			pdfCheckboxCheckNone(programParagraph, langIndicator);
		}

		// Health care coverage
		programParagraph.add(this.getDisplayText(80134, langIndicator));
		String[] backDatedMonths = dateRoutine.getBackDatedMonths(fwDate.getDate());
		final String BACK_DT_MONTH_TXT = this.getDisplayText(1800078, langIndicator);
		if (!AppConstants.ZERO.equals(rmcInPrflCargo.getIndv_fma_rqst_ind()))
		{
			pdfCheckboxCheckYes(programParagraph, langIndicator);
			// Check if back dated month questions is selected or 3. If its 3
			// then dont select any back dated months
			String backdt_med_app = rmcInPrflCargo.getBackdt_med_app();
			if ("1".equalsIgnoreCase(backdt_med_app) || "Y".equalsIgnoreCase(backdt_med_app)) {
				if (rmcInPrflCargo.getMa_backdt_mo_1() != null) {
					programParagraph.add(BACK_DT_MONTH_TXT + " 1 " + backDatedMonths[0]);
					pdfCheckboxCheckYes(programParagraph, langIndicator);
				} else {
					programParagraph.add(BACK_DT_MONTH_TXT + " 1 " + backDatedMonths[0]);
					pdfCheckboxCheckNo(programParagraph, langIndicator);
				}

				if (rmcInPrflCargo.getMa_backdt_mo_2() != null) {
					programParagraph.add(BACK_DT_MONTH_TXT + " 2 " + backDatedMonths[1]);
					pdfCheckboxCheckYes(programParagraph, langIndicator);
				} else {
					programParagraph.add(BACK_DT_MONTH_TXT + " 2 " + backDatedMonths[1]);
					pdfCheckboxCheckNo(programParagraph, langIndicator);
				}
				if (!isRetroEnabled) {
					if (rmcInPrflCargo.getMa_backdt_mo_3() != null) {
						programParagraph.add(BACK_DT_MONTH_TXT + " 3 " + backDatedMonths[2]);
						pdfCheckboxCheckYes(programParagraph, langIndicator);
					} else {
						programParagraph.add(BACK_DT_MONTH_TXT + " 3 " + backDatedMonths[2]);
						pdfCheckboxCheckNo(programParagraph, langIndicator);
					}
				}
			}
		}
		else
		{
			pdfCheckboxCheckNo(programParagraph, langIndicator);
		}
		
		// Family Planning
		programParagraph.add(this.getDisplayText(1800087, langIndicator));
		if (!AppConstants.ZERO.equals(rmcInPrflCargo.getIndv_fpp_pgm_sw()))
		{
			pdfCheckboxCheckYes(programParagraph, langIndicator);
			// Check if back dated month questions is selected or 3. If its 3
			// then dont select any back dated months
			String backdt_fpp_app = rmcInPrflCargo.getBackdt_fpp_app_sw();
			if ("1".equalsIgnoreCase(backdt_fpp_app) || "Y".equalsIgnoreCase(backdt_fpp_app)) {
				if (rmcInPrflCargo.getMa_backdt_mo_1() != null) {
					programParagraph.add(BACK_DT_MONTH_TXT + " 1 " + backDatedMonths[0]);
					pdfCheckboxCheckYes(programParagraph, langIndicator);
				} else {
					programParagraph.add(BACK_DT_MONTH_TXT + " 1 " + backDatedMonths[0]);
					pdfCheckboxCheckNo(programParagraph, langIndicator);
				}

				if (rmcInPrflCargo.getMa_backdt_mo_2() != null) {
					programParagraph.add(BACK_DT_MONTH_TXT + " 2 " + backDatedMonths[1]);
					pdfCheckboxCheckYes(programParagraph, langIndicator);
				} else {
					programParagraph.add(BACK_DT_MONTH_TXT + " 2 " + backDatedMonths[1]);
					pdfCheckboxCheckNo(programParagraph, langIndicator);
				}
				if (!isRetroEnabled) {
					if (rmcInPrflCargo.getMa_backdt_mo_3() != null) {
						programParagraph.add(BACK_DT_MONTH_TXT + " 3 " + backDatedMonths[2]);
						pdfCheckboxCheckYes(programParagraph, langIndicator);
					} else {
						programParagraph.add(BACK_DT_MONTH_TXT + " 3 " + backDatedMonths[2]);
						pdfCheckboxCheckNo(programParagraph, langIndicator);
					}
				}
			}
		}
		else
		{
			pdfCheckboxCheckNo(programParagraph, langIndicator);
		}

		// Cash Assistance
		programParagraph.add(this.getDisplayText(80187, langIndicator));
		if (AppConstants.ZERO.equals(rmcInPrflCargo.getIndv_hc_rqst_ind()))
		{
			pdfCheckboxCheckNo(programParagraph, langIndicator);
		}
		else if(ONE.equalsIgnoreCase(rmcInPrflCargo.getIndv_hc_rqst_ind()))
		{
			pdfCheckboxCheckYes(programParagraph, langIndicator);
		}
		else
		{
			pdfCheckboxCheckNone(programParagraph, langIndicator);
		}

		// Medicare
		programParagraph.add(this.getDisplayText(98710, langIndicator));
		if (AppConstants.ZERO.equals(rmcInPrflCargo.getIndv_fpw_rqst_ind()))
		{
			pdfCheckboxCheckNo(programParagraph, langIndicator);
		}
		else if (ONE.equalsIgnoreCase(rmcInPrflCargo.getIndv_fpw_rqst_ind()))
		{
			pdfCheckboxCheckYes(programParagraph, langIndicator);
		}
		else
		{
			pdfCheckboxCheckNone(programParagraph, langIndicator);
		}

		cell.addElement(programParagraph);
		return cell;
	}

	private void pdfCheckboxCheckNone(Paragraph programParagraph, String langIndicator) throws Exception
	{
		IConfiguration configuration = (IConfiguration) FwServiceFactory.getInstance().create(IConfiguration.class);
		String imageFolder = configuration.getEnvironmentProperty(FwConstants.IMAGE_FOLDER);
		Image unCheckedImg = Image.getInstance(imageFolder + "/checkbox_unchecked.png");
		unCheckedImg.scalePercent(60);

		final String YES = this.getDisplayText(30713, langIndicator);
		final String NO = this.getDisplayText(30714, langIndicator);

		programParagraph.add(new Chunk(" "));
		programParagraph.add(YES);
		programParagraph.add(new Chunk(unCheckedImg, 0, 0, true));
		programParagraph.add(new Chunk(" "));
		programParagraph.add(NO);
		programParagraph.add(new Chunk(unCheckedImg, 0, 0, true));
		programParagraph.add(new Chunk(AppConstants.NEWLINE));
	}

	/**
	 * Gathers list of programs recently selected for an individual profile (Program Add)
	 * @param rmcInPrflCargo
	 * @return
	 */
	private String prepareProgramSelected(RMC_IN_PRFL_Cargo rmcInPrflCargo,String langIndicator, ArrayList<String> previousProgramCodes) {
		
		StringBuilder programRequestedBuilder = null;
		boolean programSelected = false;
		
		
		
		if (rmcInPrflCargo != null) {
			programRequestedBuilder = new StringBuilder();
			if (!AppConstants.ZERO.equals(rmcInPrflCargo.getIndv_fs_rqst_ind()) && !previousProgramCodes.contains("FS")) {
				programSelected = true;
				programRequestedBuilder.append("\u2022").append(this.getDisplayText(32114,langIndicator)).append(AppConstants.NEWLINE);//SNAP
			}
			if (!AppConstants.ZERO.equals(rmcInPrflCargo.getIndv_hc_rqst_ind()) && !previousProgramCodes.contains("TF")) {
			    	programSelected = true;
				programRequestedBuilder.append("\u2022").append(this.getDisplayText(80187,langIndicator)).append(AppConstants.NEWLINE);//Cash Assistance
			}
			if (!AppConstants.ZERO.equals(rmcInPrflCargo.getIndv_fma_rqst_ind()) && !previousProgramCodes.contains("MA")&& !previousProgramCodes.contains("ME")&& !previousProgramCodes.contains("HB")) {
			    	programSelected = true;
				programRequestedBuilder.append("\u2022").append(this.getDisplayText(80134,langIndicator)).append(AppConstants.NEWLINE);//Healthcare coverage
			}
			if (!AppConstants.ZERO.equals(rmcInPrflCargo.getIndv_fpp_pgm_sw()) && !previousProgramCodes.contains("MA")&& !previousProgramCodes.contains("FP")) {
		    	programSelected = true;
			programRequestedBuilder.append("\u2022").append(this.getDisplayText(1800078,langIndicator)).append(AppConstants.NEWLINE);//Family planning
			}
			if (!AppConstants.ZERO.equals(rmcInPrflCargo.getIndv_fpw_rqst_ind()) && !previousProgramCodes.contains("MC")) {
			    	programSelected = true;
				programRequestedBuilder.append("\u2022").append(this.getDisplayText(98710,langIndicator)).append(AppConstants.NEWLINE);//Medicare
			}
			if(!programSelected) {
			    programRequestedBuilder.append(this.getDisplayText(36128,langIndicator)).append(AppConstants.NEWLINE);
			}
		} else {
			programRequestedBuilder = new StringBuilder().append(AppConstants.NEWLINE);
		}				
		return programRequestedBuilder.toString();
	}

	private void addProjDischargeDate(String nA_Value, RMC_IN_PRFL_Cargo prlfCargo, PdfPTable questionTable) {

			// Projected Discharge Dt
			log(ILog.DEBUG, "Inside PrintRMCPdf.addProjDischargeDate() method with appNum :" + prlfCargo.getApp_num());
			PdfPCell p117bl = this.getPdfPCell(
					this.getDisplayText(180017047, langIndicator),
					fRegularTextBold, langIndicator);
			errorSectionHeader =  this.getDisplayText( 180017047, langIndicator);
			p117bl.setGrayFill(0.8f);
			String projDischargeDtValue = AppConstants.EMPTY_STRING;
			
			String AppconstantHighDt = new SimpleDateFormat("MM/dd/yyyy")
					.format(dateRoutine
							.getDateFromTimeStamp(AppConstants.HIGH_DATE));

			if(prlfCargo.getProj_discharge_dt() != null){
				 projDischargeDtValue = new SimpleDateFormat("MM/dd/yyyy")
						.format(dateRoutine.getDateFromTimeStamp(prlfCargo.getProj_discharge_dt()));

				if (projDischargeDtValue.equals(AppconstantHighDt)) {
					projDischargeDtValue = nA_Value;
				}
			}
			questionTable.addCell(p117bl);
			questionTable.addCell(new Paragraph(projDischargeDtValue,
					fRegularTextTen));

		      FwException fe = createFwException(getClass().getName(), "addProjDischargeDate", e);
