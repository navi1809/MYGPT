	/**

	public void buildByteArraySecurePdf(FwTransaction txnBean) {
		Map request = txnBean.getRequest();
		Map pdfMap = (Map) request.get(FwConstants.PDF_MAP);
		request.put(FwConstants.PDF_BYTE_ARRAY, buildByteArray(pdfMap));
		txnBean.setRequest(request);

	}

	/**
	 * @param txnBean
	 */
	public void buildErrorSecurePdf(FwTransaction txnBean) {
		Map request = txnBean.getRequest();
		request.put(FwConstants.PDF_BYTE_ARRAY, renderSystemErrorPDF(txnBean));
	}

	public ByteArrayOutputStream renderSystemErrorPDF(FwTransaction txnBean) {
		String appNumber=null;
		try {
			Map request = txnBean.getRequest();
			ba = new ByteArrayOutputStream();
			doc = new Document();
			langIndicator = (String) request.get(AppConstants.LANGUAGE);
			docWriter = PdfWriter.getInstance(doc, ba);
			PdfDocEventsForAFB events = new PdfDocEventsForAFB(langIndicator);
			docWriter.setPageEvent(events);
			doc.setPageSize(PageSize.LETTER);
			doc.setMargins(36, 36, 48, 36);

			doc.open();
			Paragraph pNum = new Paragraph(AppConstants.EMPTY_STRING,
					fRegularTextBold);
			appNumber = (String) request.get(AppConstants.APP_NUMBER);
			appNumber = paddingNumbers(appNumber, 10);
			pNum.add(new Paragraph("\n"));
			MmcAppNum = appNumber;
			//pNum.add(ipage.getDisplayText(30608, langIndicator));
			//ILIES-209924
			if(appNumber != null && !appNumber.isEmpty()){
				String errorMessage = getDisplayText(1700011, langIndicator);
				//pNum.add(errorMessage);
				pNum.add(errorMessage);
				pNum.add(FwConstants.SPACE);
				pNum.add(appNumber);
				pNum.add(FwConstants.DOT);
				pNum.add(new Paragraph("\n"));
				pNum.add(getDisplayText(1700012, langIndicator));
			}else{
				pNum.add(getDisplayText(1700013, langIndicator));
			}
			pNum.add(FwConstants.DOT);
			//pNum.add(appNumber);
			//pNum.add(FwConstants.DOT);
			doc.add(pNum);
			doc.add(new Paragraph(AppConstants.NEWLINE));
			//return ba;
		} catch (FwException fe) {
			
			StringWriter sw = new StringWriter();
			sw.append("FwException.renderSystemErrorPDF").append("/appNumber:").append(appNumber).append("\n");
			logger.log(this.getClass(), ILog.ERROR, sw.toString());
			
			FwWrappedException we = new FwWrappedException();

			if (doc != null) {
				doc.close();
			}
			if (docWriter != null) {
				docWriter.close();
			}
		
		if(insertPDFWaterMark){
			PdfReader reader = null;
			try {

				 int n = reader.getNumberOfPages();
			        PdfStamper stamper = new PdfStamper(reader, ba);
			        Font f = new Font(30);
			        Phrase p = new Phrase("ERROR - Missing Data on PDF", f);
			        PdfGState gs1 = new PdfGState();
			        gs1.setFillOpacity(0.5f);
		            PdfContentByte over;
			        Rectangle pagesize;
			        float x, y;
			        for (int i = 1; i <= n; i++) {
			            pagesize = reader.getPageSizeWithRotation(i);
			            x = (pagesize.left() + pagesize.right()) / 2;
			            y = (pagesize.top() + pagesize.bottom()) / 2;
			            over = stamper.getOverContent(i);
			            over.saveState();
			            over.setGState(gs1);		          
			            ColumnText.showTextAligned(over, Element.ALIGN_CENTER, p, 280, 390, 45);         
	 		            over.restoreState();
			        }
			        stamper.close();
			        reader.close();
			      //  return ba;
			} catch (Exception e) {
				log(ILog.ERROR,"PrintAFBPdf::buildByte Array ::Exception ::: MmcAppNum "+ MmcAppNum + "  Exception: "+ e.getMessage() , e);
				FwException fe = new FwException();
				fe.setClassID(this.getClass().getName());
				fe.setMethodID("buildByte Array ::Exception");
				fe.setExceptionType(FwConstants.EXP_TYP_APPLICATION);
				fe.setExceptionText(String.valueOf(e));
				fe.setStackTraceValue(FwExceptionManager.getStackTraceValue(e));			
		        writeExceptionRecord(fe);	   
			}
		}
		
		//return ba;
	}
		return ba;
	}

	/**
	 * Displays the inputed textID as it's proper String output; substitute array populates each ~ in the String as necessary.
	 * @param textID Text ID from the Database to put populated on-screen.
	 * @param languageIndicator String representing the language which needs to be used.
	 * @param substitute String Array filled with the parameters needed to replace the ~ keys in the textID String.
	 * @return String representing the textID with replaced ~ values.
	 */
	public String getDisplayText(int textID, String languageIndicator, String[] substitute){
		String text = ipage.getDisplayText(textID, languageIndicator);
		if (AppConstants.LANGUAGE_CD_ARABIC.equalsIgnoreCase(languageIndicator)) {
			text = this.convertToArabic(text);
		}
		
		if(null != substitute && substitute.length > 0){
			for(int i = 0; i < substitute.length; i++){
				if(substitute[i]!=null&& !substitute[i].isEmpty()){
					text = text.replaceFirst("~", substitute[i]);
				}
			}
		}
		
		return text;
	}

	/**
	 * @param str
	 * @return
	 */
	public String convertToArabic(String str) {
		final String[] inputs = { "&#x", "&#", "\\u", "U+", "#x", "#" };
		StringBuffer result = new StringBuffer();

		for (int i = 0; i < inputs.length; i++) {
			int radix;
			int foundIndex;
			int startIndex = 0;
			final int STR_LENGTH = str.length();
			final String input = inputs[i];
			final int input_LENGTH = input.length();

			if ("&#".equals(input) || "#".equals(input)) {
				radix = 10;
			} else {
				radix = 16;
			}

			while (startIndex < STR_LENGTH) {
				foundIndex = str.indexOf(input, startIndex);

				if (foundIndex == -1) {
					result.append(str.substring(startIndex));
					break;
				}

				result.append(str.substring(startIndex, foundIndex));
				if ("\\u".equals(input) || "U+".equals(input)) {
					startIndex = foundIndex + 6;
					if (startIndex > str.length())
						startIndex = -1; // for invalid Unicode escape sequences
				} else {
					startIndex = str.indexOf(";", foundIndex);
				}

				if (startIndex == -1) {
					result.append(str.substring(foundIndex));
					break;
				}

				String tok = str.substring(foundIndex + input_LENGTH,
						startIndex);

				try {
					result.append((char) Integer.parseInt(tok, radix));
				} catch (NumberFormatException nfe) {
					try {
						if ("\\u".equals(input) || "U+".equals(input)) {
							result.append(input + tok);
						} else {
							result.append(input + tok + str.charAt(startIndex));
						}
					} catch (StringIndexOutOfBoundsException sioobe) {
						result.append(input + tok);
					}
				}

				if (!("\\u".equals(input)) && !("U+".equals(input))) {
					startIndex++;
				}
			}

			str = result.toString();
			result.setLength(0);
		}
		return str;
	}
