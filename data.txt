	private void addBulletPointString(Document doc, int textid, String text, int innerTabs, boolean addEnter) throws DocumentException {
		try {

			List<String> listText = new ArrayList<String>();
			if (textid == 12019 || textid == 12035 || textid == 12057 || textid == 18001176) {
				String data[] = text.split("</li>");
				for (String temp : data) {
					listText.add(temp);
				}
			} else {
				listText.add(text);
			}
			Font zapfdingbats = new Font();
			Chunk bullet = new Chunk("\u2022", zapfdingbats);
			Paragraph agrParag = new Paragraph(AppConstants.EMPTY_STRING, fRegularTextTen);
			int ident = innerTabs * 10;
			agrParag.setIndentationLeft(ident);
			for (String bString : listText) {
				bString = cleanHTML(bString);
				if (bString != null && !bString.isEmpty()) {
					for (int i = 0; i < innerTabs; i++) {
						agrParag.add("\t");
					}
					agrParag.add(bullet);
					agrParag.add(AppConstants.SPACE);
					agrParag.add(bString);

					if (addEnter) {
						agrParag.add(AppConstants.NEWLINE);
					}
				}
			}
			doc.add(agrParag);

		} catch (Exception e) {
			StringWriter sw = new StringWriter();

	private String tableCleanText(int id) {

		}  else if (id == 12005 || id == 12007 || id == 12009 ){
			text = text.replaceAll("&nbsp;", " ");
		}

	private String cleanHTML(String text) {

		if (text != null && !text.isEmpty()) {
			StringBuilder textC = new StringBuilder(text);

			while (text.contains("<")) {
				int a = text.indexOf("<");
				int b = text.indexOf(">", a) + 1;
				try {
					textC.delete(a, b);
					text = textC.toString();
				} catch (Exception e) {
					logger.log(this.getClass(), ILog.ERROR, "Exception while cleaning "+text);
				}
			}
			text = text.trim();
		}
		return text;
	}

	private List getUniqueNamesList(List namesList) {
		
		List uniqueNameList =  new ArrayList();
		if (namesList != null && namesList.size() > 0) {
			
			Map uniqueNamesMap = new HashMap();
			for (int i = 0; i < namesList.size(); i++) {
				
				Map nameMap = (Map) namesList.get(i);
				String indvSeqNum = (String) nameMap.get("INDV_SEQ_NUM");
				
				if (uniqueNamesMap.containsKey(indvSeqNum)) {
					String firstName = (String) nameMap.get("FST_NAM");
					String lastName = (String) nameMap.get("LAST_NAM");
					
					if (firstName != null && lastName != null) {
						uniqueNamesMap.put(indvSeqNum, nameMap);
					}
					
				} else {
					uniqueNamesMap.put(indvSeqNum, nameMap);
				}
			}
			uniqueNameList = new ArrayList(uniqueNamesMap.values());
		}
		return uniqueNameList;
	}
