	private void writeExceptionRecord(FwException fe) {
		exceptionOccured=true;
		FwWrappedException we = new FwWrappedException();
		we.setCallingClassID(this.getClass().getName());
		we.setCallingMethodID(new Throwable().getStackTrace()[0].getMethodName());			
		we.setWamsLoginID(batchNum);
		we.setUserID(jobNm);
		we.setCurrentPageID(FwConstants.BATCH);
		fe.setExceptionType(FwConstants.EXP_TYP_BATCH);
		fe.setParameterText("FwException caught MmcAppNum :: "+ MmcAppNum + " Exception: "+  fe);
		we.setFwException(fe);
		FwExceptionManager exceptionMgr = (FwExceptionManager) FwServiceFactory.getInstance().create(IException.class);
		exceptionMgr.log(we);		
	}

private void insertMissingDataMessage() throws DocumentException  {
		
		doc.add(new Paragraph("\n"));
		try {
			this.getParagraphs(doc, errorSectionHeader, fRegularTextBold);
		} catch (Exception e) {
			log(ILog.ERROR,"PrintAFBPdf::insertMissingDataMessage ::: MmcAppNum "+ MmcAppNum + "  Exception: "+ e.getMessage() , e);
			FwException fe = new FwException();
			fe.setClassID(this.getClass().getName());
			fe.setMethodID("insertMissingDataMessage");
			fe.setExceptionType(FwConstants.EXP_TYP_APPLICATION);
			fe.setExceptionText(String.valueOf(e));
			fe.setStackTraceValue(FwExceptionManager.getStackTraceValue(e));			
	        writeExceptionRecord(fe);	        
		}
		doc.add(new Paragraph("Error reading this information.",
				fRegularErrorTen));
		doc.add(new Paragraph("\n"));
		exceptionOccured = false;
		insertPDFWaterMark = true;
		errorSectionHeader ="";
	}
	
	protected void log(int level, String message, Throwable t) {
		if (logger == null) {
			logger = (ILog) FwServiceFactory.getInstance().create(ILog.class);
		}
		logger.log(this.getClass(), level, new Throwable(message, t));
	}
