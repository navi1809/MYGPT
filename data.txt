	/**
	 * @param number
	 * @param length
	 * @return
	 */
	private String paddingNumbers(String number, int length) {
		int caseLength = number.length();
		StringBuffer paddedCase = null;
		// Zero padding(Leading)
		if (caseLength < length) {
			paddedCase = new StringBuffer();
			for (int m = 0; m < (length - caseLength); m++) {
				paddedCase.append("0");
			}
			number = paddedCase.append(number).toString();
		}
		return number;
	}

	/**
	 * @param addressCargo
	 * @return
	 */
	private String addressConvert(BNFT_CASE_ADR_Cargo addressCargo) {
		StringBuffer sb = new StringBuffer();
		boolean flag = false;

		String streetNumber = "";
		if (addressCargo.getSt_num_adr() != null) {
			streetNumber = addressCargo.getSt_num_adr().trim();
		}
		String unitAddress = "";
		if (addressCargo.getUnit_adr() != null) {
			unitAddress = addressCargo.getUnit_adr().trim();
		}
		String dir = "";
		if (addressCargo.getDir_adr() != null) {
			dir = addressCargo.getDir_adr().trim();
		}
		String ruralAddress = "";
		if (addressCargo.getSt_rurl_adr() != null) {
			ruralAddress = addressCargo.getSt_rurl_adr().trim();
		}
		String sfxAddress = "";
		if (addressCargo.getSfx_adr() != null) {
			sfxAddress = addressCargo.getSfx_adr().trim();
		}
		String qdrtAddress = "";
		if (addressCargo.getQdrt_adr() != null) {
			qdrtAddress = addressCargo.getQdrt_adr().trim();
		}
		String aptAddress = "";
		if (addressCargo.getApt_adr() != null) {
			aptAddress = addressCargo.getApt_adr().trim();
		}
		String l2Address = "";
		if (addressCargo.getL2_adr() != null) {
			l2Address = addressCargo.getL2_adr().trim();
		}
		String cityAddress = "";
		if (addressCargo.getCity_adr() != null) {
			cityAddress = addressCargo.getCity_adr().trim();
		}
		String stateName = "";
		if (addressCargo.getSt_adr() != null) {
			stateName = addressCargo.getSt_adr().trim();
		}
		String zipAddress = "";
		if (addressCargo.getZip_adr() != null) {
			zipAddress = addressCargo.getZip_adr().trim();
		}

		if (streetNumber.length() > 0) {
			sb.append(streetNumber).append(FwConstants.SPACE);
			flag = true;
		}
		if (unitAddress.length() > 0) {
			sb.append(unitAddress).append(FwConstants.SPACE);
			flag = true;
		}
		if (dir.length() > 0) {
			sb.append(dir).append(FwConstants.SPACE);
			flag = true;
		}
		if (ruralAddress.length() > 0) {
			sb.append(ruralAddress).append(FwConstants.SPACE);
			flag = true;
		}
		if (sfxAddress.length() > 0) {
			sb.append(sfxAddress).append(FwConstants.SPACE);
			flag = true;
		}
		if (qdrtAddress.length() > 0) {
			sb.append(qdrtAddress).append(FwConstants.SPACE);
			flag = true;
		}
		if (aptAddress.length() > 0) {
			sb.append(aptAddress).append(FwConstants.SPACE);
			flag = true;
		}
		if (flag) {
			sb.append(AppConstants.NEWLINE);
			flag = false;
		}
		if (l2Address.length() > 0) {
			sb.append(l2Address);
			flag = true;
		}
		if (flag) {
			sb.append(AppConstants.NEWLINE);
			flag = false;
		}
		if (cityAddress.length() > 0) {
			sb.append(FwConstants.SPACE).append(cityAddress)
					.append(FwConstants.SPACE);
		}
		if (stateName.length() > 0) {
			sb.append(stateName).append(FwConstants.SPACE);
		}
		if (zipAddress.length() > 0) {
			if (zipAddress.length() > 5) {
				sb.append(zipAddress.substring(0, 5));
				sb.append(AppConstants.HYPHEN);
				sb.append(zipAddress.substring(5));
			} else {
				sb.append(zipAddress);
			}
		}
		return sb.toString();
	}
