private boolean isFemale(PeopleHandler peopleHandler, String indvSeqNum) {

				for (int j = 0; j < mswrRltColl.size(); j++)
				{
					MSWR_EXEMP_RQST_Cargo testthisCargo = mswrRltColl.getCargo(j);
					String indv_id_mswr = testthisCargo.getIndv_id();

					{
						individualExemptionsMap.put(indv_id_mswr, new ArrayList<String>());

					individualExemptionsMap.get(indv_id_mswr).add(testthisCargo.getExemp_rqst_cd());
				}
				for (Map.Entry<String, List<String>> entry : individualExemptionsMap.entrySet())
				{
					String indvId = entry.getKey();
					List<String> exemptions = entry.getValue();
					String name = "";
					String dob = null;
					String age = null;
					for (INDIVIDUAL_Custom_Cargo cargo : indvCstColl.getResults())
					{
						if (indvId.equals(cargo.getIndv_pin_num()))
						{
							name = cargo.getFst_nam() + " " + cargo.getLast_nam();
							dob = cargo.getBrth_dt();
							if (dob != null)
							{
								age = buildAgeForPDF(dob);
								name = name + "\n" + "Age: " + age;
							}
							break;
						}
					}

					PdfPCell nameCell = this.getPdfPCell(name, fRegularTextTen, langIndicator);

					tab.addCell(nameCell);
					// StringBuilder exemptionCodes = new StringBuilder();

					PdfPTable nestedTab = new PdfPTable(1);

					{
						Paragraph pRelation = new Paragraph("", fRegularTextTen);
						pRelation.add(this.getColumnValue("MSWR", 9854, exemption, langIndicator));
						pRelation.add("\n");
						PdfPCell nestedCell = this.getPdfPCellFromParagraphs(pRelation, langIndicator);

						// tab.addCell(pRelation);
					}
					tab.addCell(nestedTab);
				}
				doc.add(tab);
			}

	private PdfPCell getPdfPCellFromParagraphs(Paragraph pdfParagraph,
			String language_indicator) {

		 PdfPCell pdfPCell = null;


		if (langIndicator != null
				&& AppConstants.LANGUAGE_CD_ARABIC
						.equalsIgnoreCase(langIndicator)) {
			
			pdfPCell = new PdfPCell(pdfParagraph);

			pdfPCell.setRunDirection(PdfWriter.RUN_DIRECTION_RTL);

		} else {
			pdfPCell = new PdfPCell(pdfParagraph);
		}

		return pdfPCell;
	}

		if("0".equalsIgnoreCase(age)){
			return age = "";
		}
		else{
			return age;
		}
	}
