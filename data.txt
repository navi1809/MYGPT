SELECT
W100DP1.SYSTEM_TS() START_TS
FROM
SYSIBM.SYSDUMMY1
FOR FETCH ONLY WITH UR;

SELECT 

TC.PRD_BEG_DT,
TC.AID_CD, 
TC.ELIG_CD, 
TC.ELIG_IND,
TC.CE_EXMPT_AGE,
TC.CHLD_UNDR_14,
TC.CASE_DISA, 
TC.TRIBE_CD, 
TC.MBR_DISA,
TC.MBR_MEDI,
TC.MBR_PREG,
TC.MBR_SCHL,
TC.ET_PRTCP_CD, 
TC.ET_EXMPT_RSN_CD, 
TC.ABAWD_FG, 
TC.ABAWD_EXMPT, 
TC.NEON_CD, 
TC.SNAP_ELIG,
TC.TANF_ELIG, 
TC.RD_DATE, 
TC.ZIP_CD, 
TC.CNTY_NAME,
TC.CNTY_CD, 
TC.CE_TYPE,
TC.CE_EE,
TC.CE_AID_CD, 

COUNT(*) REC_CNT

FROM

(
SELECT 
TB.PRD_BEG_DT,
TB.HOH_UPI,
TB.APPL_UPI, 
TB.ICI,
TB.PRGM_STS, 
TB.UPI,
TB.DOB,
TB.RELSH_CD,
TB.AID_ELIG,
(CASE WHEN TB.AID_CD = 'CA'  THEN '1A) CA'
      WHEN TB.AID_CD = 'AM1' THEN '2A) AM1'
      WHEN TB.AID_CD = 'EM2' THEN '3A) EM2'
      
      WHEN TB.AID_CD = 'CH'  THEN '4A) CH'
      WHEN TB.AID_CD = 'CH1' THEN '4B) CH1'
      WHEN TB.AID_CD = 'CH4' THEN '4C) CH4'
      WHEN TB.AID_CD = 'CH5' THEN '4D) CH5'
      
      WHEN TB.AID_CD = 'AM'  THEN '5A) AM'
      WHEN TB.AID_CD = 'AM5' THEN '5B) AM5'
      
      WHEN TB.AID_CD = 'NC'  THEN '6A) NC'
      WHEN TB.AID_CD = 'TR'  THEN '7A) TR'
      
      WHEN TB.AID_CD = 'EM4' THEN '8A) EM4'
      WHEN TB.AID_CD = 'EM8' THEN '8B) EM8'
      ELSE TB.AID_CD END) AID_CD, 
      

--TB.AID_CD,
TB.ELIG_CD, 
TB.AU_NMB, 
TB.ELIG_AMT,
TB.ELIG_IND,

TB.AGE_YYMMDD,
TB.AGE_YY,
TB.CE_EXMPT_AGE,
TB.CHLD_UNDR_14,
TB.CASE_DISA, 

TB.TRIBE_CD, 
TB.MBR_DISA,
TB.MBR_MEDI,
TB.MBR_PREG,
TB.MBR_SCHL,

TB.GRS_AMT,
TB.HRS_WRKD,

TB.ET_PRTCP_CD, 
TB.ET_EXMPT_RSN_CD, 
TB.ABAWD_FG, 
TB.ABAWD_EXMPT, 

TB.NEON_CD, 
TB.SNAP_ELIG,
TB.TANF_ELIG, 
TB.RD_DATE, 
TB.ZIP_CD, 

(SELECT TQ.REFR_CD_VAL3
FROM 
W100DP1.TWNCODE_REFR TQ
WHERE TQ.REFR_CD = TB.ZIP_CD
FETCH FIRST 1 ROW ONLY
) CNTY_NAME,


TB.CNTY_CD, 

(CASE
WHEN TB.ELIG_IND = 'E' THEN
(CASE
WHEN TB.CE_EXMPT_AGE = 'YES'
THEN '1A) CE-EXEMPT-AGE'

WHEN TB.MBR_DISA = 'YES'
THEN '2A) CE-EXEMPT-DISABILITY'
WHEN TB.MBR_MEDI = 'YES'
THEN '2B) CE-EXEMPT-MEDICARE'
WHEN TB.MBR_PREG = 'YES'
THEN '2C) CE-EXEMPT-PREGNANCY'
WHEN TB.MBR_SCHL = 'YES'
THEN '2D) CE-EXEMPT-SCHOOL'
WHEN TB.POST_PART = 'YES'
THEN '2E) CE-EXEMPT-POSTPARTUM'
WHEN TB.TRIBE_CD = 'YES'
THEN '2F) CE-EXEMPT-TRIBE'

WHEN TB.CHLD_UNDR_14 = 'YES'
THEN '2G) CE-EXEMPT-CASE-CHILD-UNDR-14'
WHEN TB.CASE_DISA = 'YES'
THEN '2H) CE-EXEMPT-CASE-DISABILED-MEMBER'

WHEN (TB.ET_PRTCP_CD = 'M' OR TB.ABAWD_FG IN ('Y','W','R'))
THEN '2I) CE-EXEMPT-SNAP-ABAWD'
WHEN (TB.TANF_ELIG = '1'   AND TB.NEON_CD IN ('E','G','I','K','R','S','Z'))
THEN '2J) CE-EXEMPT-TANF-E&T'

WHEN (TB.ELIG_AMT >= 580)
THEN '3A) CE-COMPLY-INCOME >= $580'
WHEN (TB.HRS_WRKD >= 80)
THEN '3B) CE-COMPLY-HOURS >= 80'

ELSE '4A) CE-FAIL'
END)
ELSE
'9A) CE-N/A' END) CE_TYPE,

(CASE WHEN TB.AID_CD IN ('AM1','CA','EM2') 
THEN

(CASE
WHEN TB.ELIG_IND = 'E' THEN

        (CASE
        WHEN (TB.CE_EXMPT_AGE = 'YES' OR
              TB.MBR_DISA = 'YES' OR
              TB.MBR_MEDI = 'YES' OR
              TB.MBR_PREG = 'YES' OR
              TB.MBR_SCHL = 'YES' OR
              TB.POST_PART = 'YES' OR
              TB.TRIBE_CD = 'YES' OR
              TB.CHLD_UNDR_14 = 'YES' OR
              TB.CASE_DISA = 'YES' OR
              (TB.ET_PRTCP_CD = 'M' OR TB.ABAWD_FG IN ('Y','W','R')) OR
              (TB.TANF_ELIG = '1' AND TB.NEON_CD IN ('E','G','I','K','R','S','Z'))
             ) 
        THEN '1) CE Exempted' 
        WHEN (TB.ELIG_AMT >= 580 OR
              TB.HRS_WRKD >= 80)
        THEN '2) CE Comply'
        ELSE '3) CE Fail'
        END) 
ELSE '5) N/A' END) 

ELSE '4) CE-Exclusion'
END) CE_EE,

(CASE WHEN TB.AID_CD IN ('AM1','CA','EM2') THEN '1) Expanded'
ELSE '2) Regular' END) CE_AID_CD

FROM

(
SELECT
TA.PRD_BEG_DT,
TA.HOH_UPI,
TA.APPL_UPI,
TA.ICI,
TA.PRGM_STS, 
TA.UPI,
TA.DOB,
TA.RELSH_CD,

TRIM(TA.ET_PRTCP_CD) ET_PRTCP_CD, 
TRIM(TA.ET_EXMPT_RSN_CD) ET_EXMPT_RSN_CD, 
TRIM(TA.ABAWD_FG) ABAWD_FG, 
TRIM(TA.ABAWD_EXMPT) ABAWD_EXMPT,
TRIM(TA.NEON_CD) NEON_CD, 
TA.SNAP_ELIG,
TA.TANF_ELIG, 
TA.RD_DATE, 

SUBSTR(TA.ADDR_DATA,1,5) ZIP_CD, 
SUBSTR(TA.ADDR_DATA,7,2) CNTY_CD, 

TA.AID_ELIG, 

(CASE WHEN LENGTH(TRIM(TA.AID_ELIG)) > 0
THEN TRIM(
SUBSTR(TA.AID_ELIG,1,(LOCATE('-',TA.AID_ELIG))-1))
ELSE '' END) AID_CD,  

(CASE WHEN LENGTH(TRIM(TA.AID_ELIG)) > 0
THEN TRIM(
SUBSTR(TA.AID_ELIG,(LOCATE('-',TA.AID_ELIG))+1,1))
ELSE '' END) ELIG_CD, 

(CASE WHEN LENGTH(TRIM(TA.AID_ELIG)) > 0
THEN TRIM(
SUBSTR(TA.AID_ELIG,(LOCATE('-',TA.AID_ELIG))+3,2))
ELSE '' END) AU_NMB, 

CAST(
(CASE WHEN LENGTH(TRIM(TA.AID_ELIG)) > 0
THEN TRIM(
SUBSTR(TA.AID_ELIG,(LOCATE('-',TA.AID_ELIG))+10))
ELSE '0.00' END) AS DECIMAL(8,2))  ELIG_AMT,   

(CASE WHEN 
(CASE WHEN LENGTH(TRIM(TA.AID_ELIG)) > 0
THEN TRIM(
SUBSTR(TA.AID_ELIG,(LOCATE('-',TA.AID_ELIG))+1,1))
ELSE '' END) NOT IN ('',' ','3','7','9') THEN 'E'
ELSE 'I' END) ELIG_IND,

(CASE WHEN 
(CASE WHEN LENGTH(TRIM(TA.AID_ELIG)) > 0
THEN TRIM(
SUBSTR(TA.AID_ELIG,(LOCATE('-',TA.AID_ELIG))+1))
ELSE '' END) IN ('G') THEN 'YES'
ELSE 'NO' END) POST_PART, 

TA.AGE_YYMMDD,
TA.AGE_YY,
TA.CE_EXMPT_AGE,
TA.CHLD_UNDR_14,
TA.CASE_DISA,

TA.TRIBE_CD,
TA.MBR_DISA,
TA.MBR_MEDI,
TA.MBR_PREG,
TA.MBR_SCHL,

TA.GRS_AMT,
TA.HRS_WRKD

FROM

(

SELECT
T1.CLNDR_DT           PRD_BEG_DT,
LAST_DAY(T1.CLNDR_DT) PRD_END_DT,
T5.HOH_UPI,
T5.APPL_UPI,
T2.ICI,
T2.PRGM_CASE_STS PRGM_STS, 
T3.UPI,
T3.DOB,
T3.ETHNC_CD,
T6.RELSH_CD,

T4.ET_PRTCP_CD, 
T4.ET_EXMPT_RSN_CD, 
T4.ABAWD_FG, 
T4.AFDC_DUPL_ASST_IND ABAWD_EXMPT, 
T4.NEON_CD, 

COALESCE(
(
SELECT
TF.ELIG_CD
FROM
W100DP1.TWNELIG_VERSION TE,
W100DP1.TWNELG_VER_MBR  TF
WHERE TE.ICI = T2.ICI
  AND TE.PRGM_CASE_TYPE = 'FS'
  AND TE.BNFT_YR = YEAR(T1.CLNDR_DT)
  AND TE.BNFT_MTH = MONTH(T1.CLNDR_DT)
  AND TE.CASE_RSLT_CD = 'E'
  AND TE.LST_POSTD_VER_IND = 'Y'
  AND TE.ICI = TF.ICI
  AND TE.PRGM_CASE_TYPE = TF.PRGM_CASE_TYPE
  AND TE.BNFT_YR = TF.BNFT_YR
  AND TE.BNFT_MTH = TF.BNFT_MTH
  AND TE.VER_NMB = TF.VER_NMB
  AND TF.UPI = T3.UPI 
  --AND TF.ELIG_CD = '1'
),'')  SNAP_ELIG,

COALESCE(
(
SELECT
TF.ELIG_CD
FROM
W100DP1.TWNELIG_VERSION TE,
W100DP1.TWNELG_VER_MBR  TF
WHERE TE.ICI = T2.ICI
  AND TE.PRGM_CASE_TYPE = 'TN'
  AND TE.BNFT_YR = YEAR(T1.CLNDR_DT)
  AND TE.BNFT_MTH = MONTH(T1.CLNDR_DT)
  AND TE.CASE_RSLT_CD = 'E'
  AND TE.LST_POSTD_VER_IND = 'Y'
  AND TE.ICI = TF.ICI
  AND TE.PRGM_CASE_TYPE = TF.PRGM_CASE_TYPE
  AND TE.BNFT_YR = TF.BNFT_YR
  AND TE.BNFT_MTH = TF.BNFT_MTH
  AND TE.VER_NMB = TF.VER_NMB
  AND TF.UPI = T3.UPI 
  AND TF.ELIG_CD = '1'
),'')  TANF_ELIG,

(SELECT TX.NEXT_REDET_DT
FROM
W100DP1.TWNREDET_RCRT_DTLS TX
WHERE TX.ICI  = T2.ICI
  AND TX.PRGM_CASE_TYPE = 'AF'
  AND TX.NEXT_REDET_DT  >= LAST_DAY(T1.CLNDR_DT)
ORDER BY TX.NEXT_REDET_DT DESC
FETCH FIRST 1 ROW ONLY
) RD_DATE, 

(SELECT SUBSTR(TX.ZIP_CD,1,5) || '@' || TX.RSDN_CNTY_CD
FROM 
W100DP1.TWNPERSON_ADRS_HST TX
WHERE TX.UPI = T5.APPL_UPI
  AND TX.ADR_TYPE  IN ('CR', 'RA')
  AND TX.PRD_BEG_TS = (SELECT MAX(TY.PRD_BEG_TS) 
                       FROM 
                       W100DP1.TWNPERSON_ADRS_HST TY
                          WHERE TY.UPI = TX.UPI
                            AND TY.ADR_TYPE IN ('CR','RA')
                            AND LAST_DAY(T1.CLNDR_DT) BETWEEN
                                DATE(TY.PRD_BEG_TS) AND TY.PRD_END_DT)
) ADDR_DATA, 
   

COALESCE(
(
SELECT
(TF.AID_CD)  || '-' || 
(TF.ELIG_CD) || '-' ||

COALESCE(
( 
SELECT 

(
(CASE WHEN LENGTH(TRIM(T11.AU_STS)) = 0 THEN
CHAR(T11.AU_NMB) ELSE T11.AU_STS END) 

|| '-' || 
CHAR(T11.TOTAL_INCOME)
) 
--T31.TOTAL_INCOME
FROM 
W100DP1.TWNASST_UNIT      T11, 
W100DP1.TWNASST_UNIT_BDGT T21,
W100DP1.TWNASST_UNIT_MBR  T31
WHERE T11.ICI            = TE.ICI
  AND T11.PRGM_CASE_TYPE = 'AF'
  AND T11.BNFT_YR        = TE.BNFT_YR
  AND T11.BNFT_MTH       = TE.BNFT_MTH
  AND T11.VER_NMB        = TE.VER_NMB
  AND T11.TARGET_UPI     = TF.UPI
  AND T11.ICI            = T21.ICI
  AND T11.PRGM_CASE_TYPE = T21.PRGM_CASE_TYPE
  AND T11.BNFT_YR        = T21.BNFT_YR  
  AND T11.BNFT_MTH       = T21.BNFT_MTH
  AND T11.VER_NMB        = T21.VER_NMB
  AND T11.AU_NMB         = T21.AU_NMB
  AND T11.ICI            = T31.ICI
  AND T11.PRGM_CASE_TYPE = T31.PRGM_CASE_TYPE
  AND T11.BNFT_YR        = T31.BNFT_YR  
  AND T11.BNFT_MTH       = T31.BNFT_MTH
  AND T11.VER_NMB        = T31.VER_NMB
  AND T11.AU_NMB         = T31.AU_NMB
  AND T11.TARGET_UPI     = T31.UPI
FETCH FIRST 1 ROW ONLY
), '0000000-0.00')

FROM
W100DP1.TWNELIG_VERSION TE,
W100DP1.TWNELG_VER_MBR  TF
WHERE TE.ICI = T2.ICI
  AND TE.PRGM_CASE_TYPE = 'AF'
  AND TE.BNFT_YR = YEAR(T1.CLNDR_DT)
  AND TE.BNFT_MTH = MONTH(T1.CLNDR_DT)
  AND TE.CASE_RSLT_CD = 'E'
  AND TE.LST_POSTD_VER_IND = 'Y'
  AND TE.ICI = TF.ICI
  AND TE.PRGM_CASE_TYPE = TF.PRGM_CASE_TYPE
  AND TE.BNFT_YR = TF.BNFT_YR
  AND TE.BNFT_MTH = TF.BNFT_MTH
  AND TE.VER_NMB = TF.VER_NMB
  AND TF.UPI = T3.UPI ),'')  AID_ELIG,

DATE(T1.CLNDR_DT) - DATE(T3.DOB) AS AGE_YYMMDD,
INTEGER(FLOOR((DATE(T1.CLNDR_DT) -
DATE(T3.DOB)) / 10000)) AS AGE_YY,

(CASE
WHEN DATE(T1.CLNDR_DT) - DATE(T3.DOB) >= 190000
 AND DATE(T1.CLNDR_DT) - DATE(T3.DOB) < 650000
THEN 'NO' ELSE 'YES'
END) AS CE_EXMPT_AGE,

(SELECT 
(CASE 
WHEN LENGTH(TRIM(TP.TRIBE_CD)) = 0 
THEN 'NO'
WHEN TRIM(TP.TRIBE_CD) = '000'
THEN 'NO'
ELSE
'YES' END)
FROM
W100DP1.TWNPERSON_DTLS TP
WHERE TP.UPI = T3.UPI
) TRIBE_CD, 


(CASE WHEN EXISTS
(SELECT 1
FROM
W100DP1.TWNCASE_MEMBER_HST TP,
W100DP1.TWNMEMBER_REL      TQ,
W100DP1.TWNPERSON          TR
WHERE TP.ICI = T2.ICI
  AND TP.PRD_BEG_DT <= T1.CLNDR_DT
  AND TP.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  --AND TP.AFDC_ASST_IND = 'Y'
  AND TP.MAINT_CD <> 'D'
  AND ((TP.HH_LEFT_DT >= T1.CLNDR_DT)
    OR (TP.HH_LEFT_DT <= LAST_DAY(T1.CLNDR_DT)
    AND TP.HH_LEFT_RSN_CD IN('M','O','V','E')))

  AND TQ.ICI = T2.ICI
  AND TQ.PRD_BEG_DT <= T1.CLNDR_DT
  AND TQ.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  AND TQ.FROM_UPI = T5.HOH_UPI
  AND TQ.TO_UPI = TP.UPI
  AND TQ.MAINT_CD <> 'D'
  AND TQ.RELSH_CD NOT IN ('13','26')
  AND TP.UPI = TR.UPI
  AND DATE(T1.CLNDR_DT) - DATE(TR.DOB) < 140000 )
  THEN 'YES' ELSE 'NO'
END) AS CHLD_UNDR_14,

(CASE WHEN EXISTS
(SELECT 1
FROM
W100DP1.TWNCASE_MEMBER_HST TP,
W100DP1.TWNINCAP_DSBLTY    TQ
WHERE TP.ICI = T2.ICI
  AND TP.PRD_BEG_DT <= T1.CLNDR_DT
  AND TP.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  AND TP.AFDC_ASST_IND = 'Y'
  AND TP.MAINT_CD <> 'D'
  AND ((TP.HH_LEFT_DT >= T1.CLNDR_DT)
    OR (TP.HH_LEFT_DT <= LAST_DAY(T1.CLNDR_DT)
    AND TP.HH_LEFT_RSN_CD IN('M','O','V','E')))

  AND TQ.UPI = TP.UPI
  AND TQ.PRD_BEG_DT <= T1.CLNDR_DT
  AND TQ.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  AND TQ.INCAP_DSBL_VERF_CD IN ('Y','C')
  AND TQ.INCAP_DSBL_RSN_CD IN ('A','B','D','I','P','W') 
  AND TQ.MAINT_CD <> 'D'
)
THEN 'YES' ELSE 'NO'
END) AS CASE_DISA,

(CASE
WHEN EXISTS
(SELECT 1
FROM
W100DP1.TWNINCAP_DSBLTY TP
WHERE TP.UPI = T3.UPI
  AND TP.PRD_BEG_DT <= T1.CLNDR_DT
  AND TP.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  AND TP.INCAP_DSBL_VERF_CD IN ('Y','C')
  AND TP.INCAP_DSBL_RSN_CD IN ('A','B','D','I','P','W') 
  AND TP.MAINT_CD <> 'D'
)
THEN 'YES' ELSE 'NO'
END) AS MBR_DISA,

(CASE
WHEN EXISTS
(SELECT 1
FROM
W100DP1.TWNMEDICARE_INFO TP
WHERE TP.UPI = T3.UPI
  AND TP.PRD_BEG_DT <= T1.CLNDR_DT
  AND TP.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  AND (TP.PART_A_END_DT >= T1.CLNDR_DT OR
       TP.PART_B_END_DT >= T1.CLNDR_DT)
  AND TP.MAINT_CD <> 'D'
)
THEN 'YES' ELSE 'NO'
END) AS MBR_MEDI,

(CASE
WHEN EXISTS
(SELECT 1
FROM
W100DP1.TWNPRGNCY_DTLS TP
WHERE TP.UPI = T3.UPI
  AND TP.PRD_BEG_DT <= T1.CLNDR_DT
  AND TP.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  AND ((TP.PREG_TRMTN_DT = '9999-12-31')
   OR  (TP.PREG_TRMTN_DT >= T1.CLNDR_DT))
  AND TP.MAINT_CD <> 'D'
)
THEN 'YES' ELSE 'NO'
END) AS MBR_PREG,

(CASE
WHEN EXISTS
(SELECT 1
FROM
W100DP1.TWNSCHL_ATNDC TP
WHERE TP.UPI = T3.UPI
  AND TP.PRD_BEG_DT <= T1.CLNDR_DT
  AND TP.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  AND TP.SCHL_STS IN ('F','H')
  AND TP.MAINT_CD <> 'D'
)
THEN 'YES' ELSE 'NO'
END) AS MBR_SCHL,

COALESCE(
(SELECT
SUM(GRS_WGS_AMT)
FROM
(SELECT (
CASE
WHEN TM.OVRD_433_IND = 'N'
THEN ((TM.GRS_WGS_AMT/TM.PAID_CNT) * (
    CASE
        WHEN TM.PAY_FREQ = 'M'
        THEN 1
        WHEN TM.PAY_FREQ = 'S'
        THEN 2
        WHEN TM.PAY_FREQ = 'B'
        THEN 2.15
        WHEN TM.PAY_FREQ = 'W'
        THEN 4.3
        WHEN TM.PAY_FREQ = 'O'
        THEN 1
        WHEN TM.PAY_FREQ = 'Q'
        THEN (1/3)
        WHEN TM.PAY_FREQ = 'T'
        THEN (1/6)
        WHEN TM.PAY_FREQ = 'A'
        THEN 1
        ELSE 1
    END))
ELSE TM.GRS_WGS_AMT
END) GRS_WGS_AMT
FROM
(SELECT
TX.UPI,
TX.INCM_SQ_NMB,
TX.OVRD_433_IND,
TX.PAY_FREQ,
SUM(TY.HOURS_WRKD)  HOURS_WRKD,
SUM(TY.GRS_WGS_AMT) GRS_WGS_AMT,
COUNT(TY.PAID_DT)   PAID_CNT
FROM
W100DP1.TWNJOB_INCM_DTLS TX,
W100DP1.TWNPAY_DTLS      TY
WHERE TX.UPI = T3.UPI
  AND TX.PRD_BEG_DT <= T1.CLNDR_DT
  AND TX.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  AND TX.MAINT_CD <> 'D'
  AND TX.CHAP_CNTBL_IND = 'Y'
  AND TX.UPI = TY.UPI
  AND TX.PRD_BEG_DT = TY.PRD_BEG_DT
  AND TX.INCM_SQ_NMB = TY.INCM_SQ_NMB
  GROUP BY
      TX.UPI,
      TX.INCM_SQ_NMB,
      TX.OVRD_433_IND,
      TX.PAY_FREQ
ORDER BY TX.UPI) TM

UNION ALL

SELECT
((COALESCE(TB.MTHLY_GRS_INC_AMT, 0) + COALESCE
(TB.MTHLY_TIPS_AMT, 0) )) AS GRS_WGS_AMT
FROM
(SELECT
TX.UPI,
TX.INCM_SQ_NMB,
SUM(TX.MTHLY_GRS_INC_AMT) AS MTHLY_GRS_INC_AMT,
SUM(TX.MTHLY_TIPS_AMT)    AS MTHLY_TIPS_AMT,
SUM(TX.HRS_WRKD_PER_WK)   AS HRS_WRKD_PER_WK
FROM
W100DP1.TWNBUS_INCM_DTLS TX
WHERE TX.UPI = T3.UPI
  AND TX.PRD_BEG_DT <= T1.CLNDR_DT
  AND TX.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  AND TX.MAINT_CD <> 'D'
  AND TX.CHAP_CNTBL_IND = 'Y'
  GROUP BY TX.UPI, TX.INCM_SQ_NMB ) TB

UNION ALL

SELECT
SUM(TX.OTHR_ERND_INC_AMT) * (
CASE
    WHEN TX.OTHR_ERND_INC_FREQ = 'M'
    THEN 1
    WHEN TX.OTHR_ERND_INC_FREQ = 'S'
    THEN 2
    WHEN TX.OTHR_ERND_INC_FREQ = 'B'
    THEN 2.15
    WHEN TX.OTHR_ERND_INC_FREQ = 'W'
    THEN 4.3
    WHEN TX.OTHR_ERND_INC_FREQ = 'O'
    THEN 1
    WHEN TX.OTHR_ERND_INC_FREQ = 'Q'
    THEN (1/3)
    WHEN TX.OTHR_ERND_INC_FREQ = 'T'
    THEN (1/6)
    WHEN TX.OTHR_ERND_INC_FREQ = 'A'
    THEN 1
    ELSE 1
END) GRS_WGS_AMT
FROM
W100DP1.TWNOTH_ERND_INC_DT TX
WHERE TX.UPI = T3.UPI
  AND TX.PRD_BEG_DT <= T1.CLNDR_DT
  AND TX.PRD_END_DT >= LAST_DAY(T1.CLNDR_DT)
  AND TX.CHAP_CNTBL_IND = 'Y'
  AND TX.MAINT_CD <> 'D'
GROUP BY
TX.UPI,
TX.INCM_SQ_NMB,
TX.OTHR_ERND_INC_FREQ) TC), 0) GRS_AMT,

COALESCE(
(SELECT SUM(HOURS_WRKD)
FROM
(SELECT (CASE
WHEN TM.OVRD_433_IND = 'N'
THEN ((TM.HOURS_WRKD * 1.0/TM.PAID_CNT) * (
CASE
WHEN TM.PAY_FREQ = 'M'
THEN 1
