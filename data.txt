import java.io.IOException;
import java.nio.ByteBuffer;
import java.nio.CharBuffer;
import java.nio.charset.*;
import java.nio.file.*;
import java.util.*;
import java.util.stream.Stream;

public class FileTextArchiver {

    private static final Set<String> EXTENSIONS =
            new HashSet<String>(Arrays.asList(
                    ".al",
                    ".jsp",
                    ".js",
                    ".txt",
                    ".java"
            ));

    private static final String FILE_HEADER =
            "----------------- FILE -----------------";

    private static final String FILE_END =
            "----------------- END FILE -----------------";

    public static void main(String[] args) throws Exception {

        if (args.length < 3) {
            System.out.println("Usage:");
            System.out.println(
                    "Pack   : pack <source-folder> <archive-file>");
            System.out.println(
                    "Unpack : unpack <archive-file> <output-folder>");
            return;
        }

        String operation = args[0];
        Path inputPath = Paths.get(args[1]);
        Path outputPath = Paths.get(args[2]);

        if ("pack".equalsIgnoreCase(operation)) {
            pack(inputPath, outputPath);
        } else if ("unpack".equalsIgnoreCase(operation)) {
            unpack(inputPath, outputPath);
        } else {
            System.out.println("Invalid operation: " + operation);
        }
    }

    public static void pack(
            final Path sourceFolder,
            final Path archiveFile) throws IOException {

        if (!Files.isDirectory(sourceFolder)) {
            throw new IllegalArgumentException(
                    "Source folder does not exist: "
                            + sourceFolder.toAbsolutePath());
        }

        Path archiveParent =
                archiveFile.toAbsolutePath().getParent();

        if (archiveParent != null) {
            Files.createDirectories(archiveParent);
        }

        final StringBuilder archive = new StringBuilder();

        try (Stream<Path> paths = Files.walk(sourceFolder)) {

            paths.filter(Files::isRegularFile)
                    .filter(FileTextArchiver::hasAllowedExtension)
                    .filter(file -> !sameFile(file, archiveFile))
                    .forEach(file -> {

                        try {
                            Path relativePath =
                                    sourceFolder.relativize(file);

                            String storedPath =
                                    relativePath.toString()
                                            .replace("\\", "/");

                            DecodedFile decodedFile =
                                    readTextFile(file);

                            archive.append(FILE_HEADER)
                                   .append("\n");

                            archive.append("PATH=")
                                   .append(storedPath)
                                   .append("\n");

                            archive.append("CHARSET=")
                                   .append(decodedFile.charset.name())
                                   .append("\n");

                            archive.append("LENGTH=")
                                   .append(decodedFile.content.length())
                                   .append("\n");

                            archive.append("CONTENT-BEGIN")
                                   .append("\n");

                            /*
                             * Do not add or remove anything inside
                             * the actual file content.
                             */
                            archive.append(decodedFile.content);

                            archive.append("\n")
                                   .append(FILE_END)
                                   .append("\n");

                            System.out.println(
                                    "Archived: " + storedPath
                                    + " [" + decodedFile.charset.name() + "]");

                        } catch (IOException e) {
                            throw new RuntimeException(
                                    "Failed to archive file: " + file,
                                    e);
                        }
                    });
        }

        Files.write(
                archiveFile,
                archive.toString().getBytes(StandardCharsets.UTF_8));

        System.out.println(
                "Archive created at: "
                        + archiveFile.toAbsolutePath());
    }

    public static void unpack(
            Path archiveFile,
            Path outputFolder) throws IOException {

        if (!Files.isRegularFile(archiveFile)) {
            throw new IllegalArgumentException(
                    "Archive file does not exist: "
                            + archiveFile.toAbsolutePath());
        }

        Files.createDirectories(outputFolder);

        String archive =
                new String(
                        Files.readAllBytes(archiveFile),
                        StandardCharsets.UTF_8);

        int searchPosition = 0;
        int restoredCount = 0;

        while (true) {

            int headerPosition =
                    archive.indexOf(FILE_HEADER, searchPosition);

            if (headerPosition < 0) {
                break;
            }

            int pathStart =
                    archive.indexOf("PATH=", headerPosition);

            int pathEnd =
                    archive.indexOf('\n', pathStart);

            String relativePath =
                    archive.substring(
                            pathStart + "PATH=".length(),
                            pathEnd);

            int charsetStart =
                    archive.indexOf("CHARSET=", pathEnd);

            int charsetEnd =
                    archive.indexOf('\n', charsetStart);

            String charsetName =
                    archive.substring(
                            charsetStart + "CHARSET=".length(),
                            charsetEnd);

            int lengthStart =
                    archive.indexOf("LENGTH=", charsetEnd);

            int lengthEnd =
                    archive.indexOf('\n', lengthStart);

            int contentLength =
                    Integer.parseInt(
                            archive.substring(
                                    lengthStart + "LENGTH=".length(),
                                    lengthEnd));

            String contentMarker = "CONTENT-BEGIN\n";

            int contentStart =
                    archive.indexOf(
                            contentMarker,
                            lengthEnd)
                            + contentMarker.length();

            int contentEnd =
                    contentStart + contentLength;

            if (contentEnd > archive.length()) {
                throw new IOException(
                        "Invalid archive content length for: "
                                + relativePath);
            }

            String content =
                    archive.substring(contentStart, contentEnd);

            Path outputFile =
                    createSafeOutputPath(
                            outputFolder,
                            relativePath);

            Path parent = outputFile.getParent();

            if (parent != null) {
                Files.createDirectories(parent);
            }

            Charset originalCharset =
                    Charset.forName(charsetName);

            Files.write(
                    outputFile,
                    content.getBytes(originalCharset));

            System.out.println(
                    "Recreated: "
                            + outputFile.toAbsolutePath());

            restoredCount++;

            searchPosition = contentEnd;
        }

        System.out.println(
                "Total files recreated: " + restoredCount);

        System.out.println(
                "Files recreated in: "
                        + outputFolder.toAbsolutePath());
    }

    private static DecodedFile readTextFile(
            Path file) throws IOException {

        byte[] bytes = Files.readAllBytes(file);

        /*
         * First try UTF-8 strictly.
         */
        try {
            String content =
                    decodeStrictly(
                            bytes,
                            StandardCharsets.UTF_8);

            return new DecodedFile(
                    content,
                    StandardCharsets.UTF_8);

        } catch (CharacterCodingException e) {

            /*
             * Many older Windows project files use
             * Windows-1252 encoding.
             */
            Charset windowsCharset =
                    Charset.forName("windows-1252");

            try {
                String content =
                        decodeStrictly(
                                bytes,
                                windowsCharset);

                return new DecodedFile(
                        content,
                        windowsCharset);

            } catch (CharacterCodingException secondException) {

                /*
                 * Final fallback.
                 */
                Charset fallback =
                        StandardCharsets.ISO_8859_1;

                String content =
                        new String(bytes, fallback);

                return new DecodedFile(
                        content,
                        fallback);
            }
        }
    }

    private static String decodeStrictly(
            byte[] bytes,
            Charset charset)
            throws CharacterCodingException {

        CharsetDecoder decoder =
                charset.newDecoder();

        decoder.onMalformedInput(
                CodingErrorAction.REPORT);

        decoder.onUnmappableCharacter(
                CodingErrorAction.REPORT);

        CharBuffer result =
                decoder.decode(ByteBuffer.wrap(bytes));

        return result.toString();
    }

    private static boolean hasAllowedExtension(
            Path file) {

        String fileName =
                file.getFileName()
                    .toString()
                    .toLowerCase();

        for (String extension : EXTENSIONS) {
            if (fileName.endsWith(extension)) {
                return true;
            }
        }

        return false;
    }

    private static boolean sameFile(
            Path first,
            Path second) {

        return first.toAbsolutePath()
                    .normalize()
                    .equals(
                        second.toAbsolutePath()
                              .normalize());
    }

    private static Path createSafeOutputPath(
            Path outputFolder,
            String relativePath) throws IOException {

        Path normalizedFolder =
                outputFolder.toAbsolutePath()
                            .normalize();

        Path outputPath =
                normalizedFolder
                    .resolve(relativePath)
                    .normalize();

        if (!outputPath.startsWith(normalizedFolder)) {
            throw new IOException(
                    "Invalid path in archive: "
                            + relativePath);
        }

        return outputPath;
    }

    private static class DecodedFile {

        private final String content;
        private final Charset charset;

        private DecodedFile(
                String content,
                Charset charset) {

            this.content = content;
            this.charset = charset;
        }
    }
}
